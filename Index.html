export default async function handler(req, res) { if (req.method !== "POST") { return res.status(405).json({ reply: "Method not allowed" }); } try { const { message } = req.body; if (!message) { return res.status(400).json({ reply: "Please enter a question." }); } const response = await fetch( "https://api.groq.com/openai/v1/chat/completions", { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${process.env.GROQ_API_KEY}` }, body: JSON.stringify({ model: "llama-3.3-70b-versatile", messages: [ { role: "system", content: "You are a helpful school AI assistant. " + "Answer students clearly, politely and safely. " + "If you don't know school-specific information, " + "say that you don't know instead of making it up." }, { role: "user", content: message } ], temperature: 0.5 }) } ); const data = await response.json(); if (!response.ok) { return res.status(response.status).json({ reply: "Groq API error. Please check the server configuration." }); } const reply = data.choices?.[0]?.message?.content || "I could not generate a response."; return res.status(200).json({ reply }); } catch (error) { console.error(error); return res.status(500).json({ reply: "Server error. Please try again." }); } }
